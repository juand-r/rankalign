# Requirements for RunPod pods running gemma-4 (31B) inference / scoring.
#
# Why this exists:
#   The default rankalign requirements.txt pins transformers==4.46.2 +
#   tokenizers==0.20.3, which CANNOT load gemma-4. Installing the default
#   requirements.txt and then upgrading transformers separately breaks
#   scipy/sklearn binary compat too — the safest path is to install
#   gemma-4-compatible versions in a single resolution pass.
#
# Usage:
#   pip install --upgrade -r requirements-runpod-gemma4.txt
#   pip install --upgrade "git+https://github.com/huggingface/transformers.git"
#
# The transformers main install is a separate step because pip can't
# resolve a git+ entry the same way as a versioned one when mixed with
# other constraints in this file.

# Pinned-compatible core (these versions resolve together and don't fight transformers main):
numpy>=2.1
scipy>=1.14
pandas>=2.2
scikit-learn>=1.5

# HF stack: leave UNPINNED — let pip pick versions compatible with transformers main.
# The default rankalign requirements.txt clobbers all three with stale pins; do NOT pin here.

tokenizers
accelerate
datasets
